cmake_minimum_required(VERSION 3.20)
project(vor_async_test LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core)
add_executable(vor_async_test test.cpp ../../../src/core/VorController.cpp ../../../src/core/VorController.h ../../../src/core/VorAsyncReceiver.cpp ../../../src/core/AsyncIqDecoder.cpp ../../../src/dsp/VorReceiver.cpp ../../../src/dsp/VorIdentDecoder.cpp)
target_include_directories(vor_async_test PRIVATE ../../../src)
target_link_libraries(vor_async_test PRIVATE Qt6::Core)
find_package(Qt6 REQUIRED COMPONENTS Gui Qml Quick QuickControls2)
add_executable(vor_gui_test gui.cpp ../../../src/core/VorController.cpp ../../../src/core/VorController.h ../../../src/core/VorAsyncReceiver.cpp ../../../src/core/AsyncIqDecoder.cpp ../../../src/dsp/VorReceiver.cpp ../../../src/dsp/VorIdentDecoder.cpp)
target_include_directories(vor_gui_test PRIVATE ../../../src)
target_link_libraries(vor_gui_test PRIVATE Qt6::Core Qt6::Gui Qt6::Qml Qt6::Quick Qt6::QuickControls2)
target_compile_definitions(vor_gui_test PRIVATE VOR_TEST_QML="${CMAKE_CURRENT_SOURCE_DIR}/qml" VOR_TEST_BASE="${CMAKE_CURRENT_SOURCE_DIR}")
